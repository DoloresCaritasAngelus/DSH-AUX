	/**
	* Namespaces registered with `exposedToWeb: true` — the configuration-client
	* (Web settings API) boundary. Plugins opt in at registration; without this,
	* a namespace stays Host-only even though its owner registered it.
	* @returns the exposed namespace strings, in registration order.
	*/
	listExposed() {
		return [...this.registrations.values()].filter((r) => r.exposedToWeb === true).map((r) => r.ns);
	}
