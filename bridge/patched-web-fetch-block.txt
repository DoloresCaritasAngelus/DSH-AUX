		async execute(args, exec) {
			const input = parseFetchArgs(args);
			// dsh-aux web_fetch compat (local patch)
			let compat;
			try {
				compat = ctx.get?.("auxLlm");
			} catch {
				compat = void 0;
			}
			if (compat?.toolBridgeMode?.("web_extract") === "compat" && typeof compat.webFetchCompat === "function") {
				return compat.webFetchCompat(input, exec);
			}
			const result = await ctx.web.fetch({ url: input.url }, exec.signal);
			return {
				url: result.url,
				statusCode: result.statusCode,
				body: {
					kind: result.body.kind,
					content: result.body.content
				},
				truncated: result.truncated
			};
		},
