			for (const block of message.content) {
				if (block?.type === "image" && block.attachment !== void 0) {
					const id = String(block.attachment.attachmentId ?? "");
					const match = /^sha256:([a-f0-9]{64})$/.exec(id);
					const hash = match?.[1];
					if (hash !== void 0 && home !== void 0) {
						const real = `${home}/attachments/v1/objects/${hash.slice(0, 2)}/${hash}`;
						const ext = extOf(block.attachment.mediaType);
						const linkPath = `${real}.${ext}`;
						try {
							// Ensure the extension hard link exists (the admit bridge
							// creates it for fresh uploads; older logs may lack it).
							await link(real, linkPath).catch((e) => { if (e?.code !== "EEXIST") throw e; });
						} catch { /* keep the real path when linking fails */ }
						content.push({ type: "text", text: `[用户上传了一张图片，本地路径: ${linkPath}。请用 vision_analyze 工具（imagePath 参数）查看这张图片并回答用户的问题。]` });
						continue;
					}
				}
				content.push(block);
